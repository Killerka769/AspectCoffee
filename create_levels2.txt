from apps.users.models import UserLevel

# Delete old levels and create new ones without emoji
UserLevel.objects.all().delete()

levels = [
    ("Новичок", 0, 0, "", "#999999", 1),
    ("Любитель", 3, 0, "", "#4CAF50", 2),
    ("Кофеман", 10, 0, "", "#FF9800", 3),
    ("Эксперт", 20, 0, "", "#9C27B0", 4),
    ("Мастер", 50, 0, "", "#2196F3", 5),
    ("Легенда", 100, 0, "", "#FF6B00", 6),
]

for name, min_orders, min_spent, icon, color, sort in levels:
    UserLevel.objects.create(
        name=name,
        min_orders=min_orders,
        min_spent=min_spent,
        icon=icon,
        color=color,
        sort_order=sort,
        is_active=True,
    )
    print(f"Created: {name}")
